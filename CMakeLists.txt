cmake_minimum_required(VERSION 3.22)
project(NOXIS VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NOXIS_BUILD_APP "Build the JUCE standalone application" ON)
add_library(noxis_dsp src/synth/Engine.cpp src/state/Preset.cpp)
target_include_directories(noxis_dsp PUBLIC src)
if(MSVC)
  target_compile_options(noxis_dsp PRIVATE /W4 /permissive- /utf-8)
endif()
add_executable(noxis_render src/audio/OfflineRenderer.cpp)
target_link_libraries(noxis_render PRIVATE noxis_dsp)
enable_testing()
add_executable(noxis_tests tests/DspTests.cpp)
target_link_libraries(noxis_tests PRIVATE noxis_dsp)
add_test(NAME DSP_and_presets COMMAND noxis_tests)
add_test(NAME Offline_WAV COMMAND noxis_render ${CMAKE_CURRENT_BINARY_DIR}/noxis-demo.wav)
if(NOXIS_BUILD_APP)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/JUCE/CMakeLists.txt")
    add_subdirectory(external/JUCE EXCLUDE_FROM_ALL)
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
      GIT_TAG 8.0.16-security GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  juce_add_gui_app(NOXIS PRODUCT_NAME "NOXIS" COMPANY_NAME "NOXIS" VERSION ${PROJECT_VERSION})
  target_sources(NOXIS PRIVATE src/app/Main.cpp src/audio/AudioHost.cpp src/ui/MainComponent.cpp src/ui/Controls.cpp)
  target_include_directories(NOXIS PRIVATE src)
  target_compile_definitions(NOXIS PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="$<TARGET_PROPERTY:NOXIS,JUCE_PRODUCT_NAME>" JUCE_APPLICATION_VERSION_STRING="${PROJECT_VERSION}")
  target_link_libraries(NOXIS PRIVATE noxis_dsp juce::juce_audio_utils juce::juce_recommended_config_flags)
  if(MSVC)
    target_compile_options(NOXIS PRIVATE /utf-8 /W4)
  endif()
endif()
